# create random shift amount

OP_ASSIGN_INT v:LAST_INT v:randShifter

OP_CONTROL_JUMP_CONDITIONAL l:makePositive
OP_CONTROL_JUMP l:staySame
.label makePositive
OP_MULTIPLY_CONST_INT v:randShifter hx:00000004
OP_ASSIGN_INT v:LAST_INT v:randShifter
OP_POP_INT
OP_CONTROL_JUMP l:done
.label staySame
OP_MULTIPLY_CONST_INT v:randShifter hx:FFFFFFFC
OP_ASSIGN_INT v:LAST_INT v:randShifter
OP_POP_INT
OP_CONTROL_JUMP l:done

.label done

# apply shift and place

OP_SUB_ASSIGN_INT v:mapDataArray v:LAST_INT v:floorCounter
OP_POP_INT

# move up floor counter for next iteration
OP_INC_INT v:floorCounter 2

OP_CONTROL_JUMP l:whileLessThan799
.label whileLessThan799End
